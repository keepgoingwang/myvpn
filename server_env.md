# ============================================================
# My VPN Server Configuration
# ============================================================

# ------------------------------------------------------------
# Server
# ------------------------------------------------------------

# Shadowsocks 服务端监听端口
#
# 推荐：
#   18188  -> 普通测试
#   443    -> 如果你后续希望使用 443
#
SERVER_PORT=18188


# ------------------------------------------------------------
# Shadowsocks
# ------------------------------------------------------------

# Shadowsocks 加密方式
#
# 当前使用：
# aes-256-gcm
#
SS_METHOD=aes-256-gcm

# Shadowsocks 密码
#
# 如果留空，install.sh 会自动生成随机密码。
#
# 第一次部署建议：
# SS_PASSWORD=
#
# 如果你自己指定：
# SS_PASSWORD=your-strong-password
#
SS_PASSWORD=

# 节点名称
#
# 会显示在 Clash / Shadowsocks 客户端中
#
NODE_NAME=US-Aliyun


# ------------------------------------------------------------
# Public Network
# ------------------------------------------------------------

# 公网 IP
#
# 留空：
#   自动通过 ifconfig.me 获取
#
# 如果服务器有多个公网 IP，
# 可以手动指定。
#
PUBLIC_IP=


# ------------------------------------------------------------
# Subscription
# ------------------------------------------------------------

# 是否启用 Nginx 订阅服务
#
# true  = 启用
# false = 不启用
#
ENABLE_SUBSCRIPTION=true


# 订阅随机路径
#
# 例如：
#
# https://example.com/8f31c9d1a2/subscribe.txt
#
# 留空：
#   自动生成
#
SUBSCRIBE_TOKEN=


# ------------------------------------------------------------
# Domain
# ------------------------------------------------------------

# 你的域名
#
# 如果没有域名：
#
# DOMAIN=
#
# 那么订阅地址会使用：
#
# http://服务器IP/随机TOKEN/subscribe.txt
#
# 如果有域名：
#
# DOMAIN=vpn.example.com
#
DOMAIN=


# ------------------------------------------------------------
# HTTPS
# ------------------------------------------------------------

# 是否启用 HTTPS
#
# false：
#   只配置 HTTP
#
# true：
#   要求 DOMAIN 已正确解析到服务器
#   并使用 Let's Encrypt / Certbot 配置 HTTPS
#
ENABLE_HTTPS=false


# Let's Encrypt 邮箱
#
# ENABLE_HTTPS=true 时使用
#
ACME_EMAIL=


# ------------------------------------------------------------
# Nginx
# ------------------------------------------------------------

# Nginx HTTP 端口
#
NGINX_HTTP_PORT=80

# Nginx HTTPS 端口
#
NGINX_HTTPS_PORT=443


# ------------------------------------------------------------
# Firewall
# ------------------------------------------------------------

# 是否自动配置 UFW
#
# true：
#   自动检查并开放：
#   SSH
#   Shadowsocks
#   HTTP
#   HTTPS
#
# false：
#   不修改防火墙
#
ENABLE_UFW=true


# SSH 端口
#
# 阿里云默认一般是 22
#
SSH_PORT=22
